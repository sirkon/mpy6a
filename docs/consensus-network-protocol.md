# Описание транспортного протокола консенсуса работающего поверх TCP.

Консенсус функционирует на транспортном протоколе работающем поверх TCP.

Каждая запись проходит в своем фрейме. Каждый фрейм имеет порядковый номер. Это необходимое условие, т.к. момент выхода
пакета может быть не привязан к хронологической последовательности запроса на передачу. Это связано с отдачей задач на
разные ядерные воркеры – из-за них и может случится рассинхрон.

Чтобы обрабатывать подобные коллизии, введен т.н. "ReorderBuffer", который берет на себя управление переупорядочиванием
– которое, к слову сказать, нужно весьма редко. Большинство пакетов практически всегда сохраняет порядок. Нарушенный
порядок это буквально одно событие на сотни или даже тысячи произведенных записей.

Для того, чтобы он работал, в каждый пакет первыми двумя байтами запекается даже не порядковый номер пакета, а кольцевой
номер:

```
0, 1, 2, …, 65534, 65535, 0, 1, …, 65535, 0, …
```

Окно переупорядочивания значительно меньше по размеру, поэтому двух байт будет достаточно. А один байт может быть
рискованным.

Протокол обладает мультиплексированием, никакой немедленной отдачи ответа на присланные данные не ожидается. Внутри
данные выглядят следующим образом:

```go
type Payload struct {
	// При отправке пакетов циклически пробегаются все хосты индекса и отправляется их локальный индекс.
	// Это сделано для подтаскивания данных по хосту, с которым у данного нет связи.
	HostInfo struct {
    ID    u8
    Index u128
	}
	Op     Operation
}

type Operation oneof {
	// Создание новой сессии.
  Create struct {
    SessionID   u128
    Compensator bytes
    Data        bytes
  }

  // Добавление новых данных в текущую сессию.
  Append struct {
  	SessionID u128
    Data       bytes
  }

  // Перезапись существующих данных в сессии приложенными.
  Overwrite struct {
  	SessionID u128
    Data       bytes
  }

  Repeat struct {
    SessionID u128
    RepeatIn  u32 // время в секундах.
  }

  Close struct {
    SessionID u128
  }

  // Когда HostID слишком долго недоступен и предлагается его изолировать.
  Isolate struct {
  	HostID u8
  }

  Compensate struct {
  	HostID u8  // сессии какого хоста компенсируются
    Count  u16 // сколько сессий компенсируются. Имеется ввиду, забираются count сессий для запуска компенсации.
  }

  // Рассылается хостом X в подключение к хосту Y чтобы показать последнему свою работоспособность.
  Pulse struct{}

  Ack struct {
    Epoch     u128 // Своя эпоха. Если вся остальная часть кластера решила, что хост недоступен,
                   // то устаревшая эпоха данного поможет им понять, что его подтверждение неактуально.

    Op oneof {
      Create u128,
      Append u128,
      Overwrite u128,
      Repeat u128,
      Close u128,
      Isolate u8,
      Compensate struct {
        HostID u8
        Count  u16
      }
    }
  }

  Nack struct {
    Epoch     u128 // Своя эпоха. Если вся остальная часть кластера решила, что хост недоступен,
                   // то устаревшая эпоха данного поможет им понять, что его подтверждение неактуально.

    Op oneof {
      Create u128,
      Append u128,
      Overwrite u128,
      Repeat u128,
      Close u128,
      Isolate u8,
      Compensate struct {
        HostID u8
        Count  u16
      }
    }

  }

  // Запрос лога с данного хоста после позиции Index.
  RequestLog struct {
  	HostID u8
    Index  u128
  }

  // Кусок данных по хосту начиная с позиции Index кодированный в каком-то виде.
  RequestedLog struct {
  	HostID u8
    Index  u128
    Final  bool
    Data   bytes
  }

  // Ответ о невозможности получения данных по RequestLog
  RequestedLogNotAvailable oneof {
 	  HostID u8
    Index  u128 // Данные из RequestLog
    Code oneof { OutOfDate, InternalError } // Слишком старая
  }

  // Запрос на получение новейшего слепка данных с хоста.
  RequestSnapshot struct {
    HostID u8
  }

  // Связанный инвариант - снапшот не удаляется, если его читает хотя бы одно зеркало.
  RequestedSnapshot struct {
  	HostID u8
    Index  u128  // Какому индексу хоста соответствует снапшот
    Final  bool  // Последний ли кусок данных снапшота.
    Data   bytes // Данные снапшота в каком-то формате
  }
}
```
